cmake_minimum_required(VERSION 3.16)
project(virtual_memory_simulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(jsoncpp CONFIG REQUIRED)
add_library(memory_core FileConfiguration.cpp Page_frame.cpp Ram_memory.cpp TLB_table.cpp Virtual_Memory_manager.cpp)
target_include_directories(memory_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(TARGET JsonCpp::JsonCpp)
  target_link_libraries(memory_core PUBLIC JsonCpp::JsonCpp)
elseif(TARGET jsoncpp_lib)
  target_link_libraries(memory_core PUBLIC jsoncpp_lib)
else()
  target_link_libraries(memory_core PUBLIC jsoncpp_static)
endif()
add_executable(memory_simulator OS_project.cpp)
target_link_libraries(memory_simulator PRIVATE memory_core)
option(BUILD_GUI "Build the optional raylib view" OFF)
if(BUILD_GUI)
  find_package(raylib 5.5 REQUIRED)
  add_executable(memory_gui gui_main.cpp Drawer.cpp)
  target_link_libraries(memory_gui PRIVATE memory_core raylib)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(memory_checks tests/core_checks.cpp)
  target_link_libraries(memory_checks PRIVATE memory_core)
  add_test(NAME memory_core_checks COMMAND memory_checks)
endif()
